# ContrattoOrologio.click – configurazione per hosting Apache / LiteSpeed (Hostinger, Aruba, SiteGround, Register.it, ecc.)
# Va caricato nella cartella principale del sito (su Hostinger: public_html), insieme a index.html.
# Il nome deve essere esattamente .htaccess (con il punto iniziale e senza .txt finale).
# Prova che funziona: aprendo https://contrattoorologio.click/fonts/ deve comparire "403 Forbidden".

# 1. Forza HTTPS e il dominio senza www
RewriteEngine On
RewriteCond %{HTTPS} off [OR]
RewriteCond %{HTTP_HOST} ^www\. [NC]
RewriteRule ^ https://contrattoorologio.click%{REQUEST_URI} [L,R=301]

# 2. HTTPS obbligatorio anche nelle visite successive (HSTS, un anno)
Header set Strict-Transport-Security "max-age=31536000; includeSubDomains"

# 3. Content Security Policy: il browser blocca qualsiasi invio di dati verso siti non elencati.
#    connect-src = gli unici indirizzi contattabili (elenco comuni + Google Analytics solo col consenso)
#    form-action 'none' = nessun modulo può essere inviato da nessuna parte
#    script-src / style-src = eseguiti solo lo script e lo stile di index.html (verificati con hash sha256)
#    Dopo ogni modifica a index.html esegui: python3 tools/csp-hash.py --write
Header set Content-Security-Policy "default-src 'none'; script-src 'sha256-v8ABIugMcyfv3hPfTSpOE7idHd121WnaHWnJSqyR9mY=' https://www.googletagmanager.com; style-src 'sha256-XEtX6KSBza7EpjloUb+jHXD33/Nz9kkFWilyp6YXjwc='; font-src 'self'; img-src 'self' data: https://*.google-analytics.com https://*.googletagmanager.com; connect-src https://cdn.jsdelivr.net https://*.google-analytics.com https://*.analytics.google.com https://*.googletagmanager.com; manifest-src 'self'; base-uri 'none'; form-action 'none'; frame-ancestors 'none'; object-src 'none'; upgrade-insecure-requests"

# 4. Altre intestazioni di sicurezza
Header set X-Content-Type-Options "nosniff"
Header set X-Frame-Options "DENY"
Header set X-XSS-Protection "0"
Header set Referrer-Policy "no-referrer"
Header set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=(), usb=(), interest-cohort=()"
Header set Cross-Origin-Opener-Policy "same-origin"

# 5. Caratteri tipografici: tipo di file corretto
AddType font/woff .woff

# 6. Niente elenco dei file della cartella
Options -Indexes
